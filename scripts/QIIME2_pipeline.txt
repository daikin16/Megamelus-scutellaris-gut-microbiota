#!/usr/bin/env bash
################################################################################
# 16S rRNA gene amplicon processing with QIIME2
# Gut microbiota of Megamelus scutellaris
# Supplementary command-line script (run 23 August 2025)
#
# QIIME2 version: ADD VERSION
#
# Sections
#    1. Activate environment
#    2. Import raw data
#    3. Quality of raw reads
#    4. Primer removal
#    5. Denoising with DADA2 (truncation length, chimera removal)
#    6. Denoising statistics and feature table summaries
#    7. Rarefaction curves (unfiltered table)
#    8. Taxonomic classification
#    9. Taxonomic filtering
#   10. Rarefaction curves (filtered table)
#   11. Taxa bar plots
#   12. Phylogenetic tree
#   13. Core diversity metrics
#   14. Alpha diversity
#   15. Beta diversity
#
# Note: commands are listed in the order they were run. Input file names
# (e.g. Manifestfrank.txt, sample-metadata.tsv) must be in the working directory.
################################################################################


################################################################################
# 1. ACTIVATE ENVIRONMENT
################################################################################

source /usr/local/bin/qiime2-activate


################################################################################
# 2. IMPORT RAW DATA
################################################################################

qiime tools import \
  --type 'SampleData[PairedEndSequencesWithQuality]' \
  --input-path Manifestfrank.txt \
  --output-path demux.qza \
  --input-format PairedEndFastqManifestPhred33V2


################################################################################
# 3. QUALITY OF RAW READS
################################################################################

qiime demux summarize \
  --i-data demux.qza \
  --o-visualization qualities.qzv


################################################################################
# 4. PRIMER REMOVAL
################################################################################

# Full primer lengths: 17 bp (forward), 20 bp (reverse)
qiime cutadapt trim-paired \
  --i-demultiplexed-sequences demux.qza \
  --p-front-f CCTAYGGGRBGCASCAG \
  --p-front-r GGACTACNNGGGTATCTAAT \
  --o-trimmed-sequences trimmed_demux.qza \
  --verbose

# Quality of reads after primer removal
qiime demux summarize \
  --i-data trimmed_demux.qza \
  --o-visualization trimmed_demux.qzv


################################################################################
# 5. DENOISING WITH DADA2 (TRUNCATION LENGTH, CHIMERA REMOVAL)
################################################################################

# 5.1 Initial attempt (reverse reads truncated at 258 bp; not used further)
# Substantial loss at Waainek B, C and D, so the truncation length was adjusted.
#qiime dada2 denoise-paired \
#  --i-demultiplexed-seqs trimmed_demux.qza \
#  --p-trunc-len-f 0 \
#  --p-trunc-len-r 258 \
#  --p-trim-left-f 0 \
#  --p-trim-left-r 0 \
#  --p-n-threads 0 \
#  --o-table table.qza \
#  --o-representative-sequences rep-seqs.qza \
#  --o-denoising-stats denoising-stats.qza \
#  --verbose

# 5.2 Final settings (reverse reads truncated at 250 bp; used for all analyses)
# 250 bp worked well without compromising quality.
qiime dada2 denoise-paired \
  --i-demultiplexed-seqs trimmed_demux.qza \
  --p-trunc-len-f 0 \
  --p-trunc-len-r 250 \
  --p-trim-left-f 0 \
  --p-trim-left-r 0 \
  --p-max-ee-f 2.0 \
  --p-max-ee-r 3.0 \
  --p-n-threads 0 \
  --o-table table-opt250.qza \
  --o-representative-sequences rep-seqs-opt250.qza \
  --o-denoising-stats denoising-stats-opt250.qza \
  --verbose


################################################################################
# 6. DENOISING STATISTICS AND FEATURE TABLE SUMMARIES
################################################################################

# 6.1 Denoising statistics
qiime metadata tabulate \
  --m-input-file denoising-stats-opt250.qza \
  --o-visualization denoising-stats-opt250.qzv

# 6.2 Feature table summary: confirm sequencing depth and ASV distribution
qiime feature-table summarize \
  --i-table table-opt250.qza \
  --m-sample-metadata-file sample-metadata.tsv \
  --o-visualization table-opt250.qzv

# 6.3 Representative sequences
qiime feature-table tabulate-seqs \
  --i-data rep-seqs-opt250.qza \
  --o-visualization rep-seqs-opt250.qzv


################################################################################
# 7. RAREFACTION CURVES (UNFILTERED TABLE)
################################################################################

qiime diversity alpha-rarefaction \
  --i-table table-opt250.qza \
  --p-max-depth 40000 \
  --p-steps 10 \
  --p-iterations 10 \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization alpha-rarefaction.qzv


################################################################################
# 8. TAXONOMIC CLASSIFICATION
################################################################################

# SILVA 138 (99% OTUs) naive Bayes classifier
qiime feature-classifier classify-sklearn \
  --i-classifier silva-138-99-nb-classifier.qza \
  --i-reads rep-seqs-opt250.qza \
  --o-classification taxonomy-opt250.qza

qiime metadata tabulate \
  --m-input-file taxonomy-opt250.qza \
  --o-visualization taxonomy-opt250.qzv


################################################################################
# 9. TAXONOMIC FILTERING
################################################################################

# Remove mitochondrial, chloroplast and unclassified sequences
qiime taxa filter-table \
  --i-table table-opt250.qza \
  --i-taxonomy taxonomy-opt250.qza \
  --p-exclude mitochondria,chloroplast,unclassified \
  --o-filtered-table filtered-table-opt250.qza

# Check the impact of filtering on the ASV count
# Filtering removed one ASV (possibly unclassified): 76 ASVs before, 75 after.
qiime feature-table summarize \
  --i-table filtered-table-opt250.qza \
  --o-visualization filtered-table-opt250.qzv


################################################################################
# 10. RAREFACTION CURVES (FILTERED TABLE)
################################################################################

# Rarefaction curves re-run with the filtered table
qiime diversity alpha-rarefaction \
  --i-table filtered-table-opt250.qza \
  --p-max-depth 40000 \
  --p-steps 10 \
  --p-iterations 10 \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization alpha-rarefaction-filtered.qzv


################################################################################
# 11. TAXA BAR PLOTS
################################################################################

qiime taxa barplot \
  --i-table filtered-table-opt250.qza \
  --i-taxonomy taxonomy-opt250.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization taxa-barplot-opt250.qzv


################################################################################
# 12. PHYLOGENETIC TREE
################################################################################

qiime phylogeny align-to-tree-mafft-fasttree \
  --i-sequences rep-seqs-opt250.qza \
  --o-alignment aligned-seqs-opt250.qza \
  --o-masked-alignment masked-seqs-opt250.qza \
  --o-tree unrooted-tree-opt250.qza \
  --o-rooted-tree rooted-tree-opt250.qza


################################################################################
# 13. CORE DIVERSITY METRICS (RAREFIED TO 40,000 READS)
################################################################################

qiime diversity core-metrics-phylogenetic \
  --i-table filtered-table-opt250.qza \
  --i-phylogeny rooted-tree-opt250.qza \
  --p-sampling-depth 40000 \
  --m-metadata-file sample-metadata.tsv \
  --output-dir diversity \
  --verbose

# Summarise the rarefied table: verify ASV retention and read counts
qiime feature-table summarize \
  --i-table diversity/rarefied_table.qza \
  --o-visualization diversity/rarefied-table-opt250.qzv


################################################################################
# 14. ALPHA DIVERSITY
################################################################################

# 14.1 Faith's phylogenetic diversity
qiime diversity alpha-group-significance \
  --i-alpha-diversity diversity/faith_pd_vector.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/faith_pd-significance.qzv \
  --verbose

# 14.2 Observed features
qiime diversity alpha-group-significance \
  --i-alpha-diversity diversity/observed_features_vector.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/observed_features-significance.qzv \
  --verbose

# 14.3 Shannon diversity
qiime diversity alpha-group-significance \
  --i-alpha-diversity diversity/shannon_vector.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/shannon-significance.qzv \
  --verbose

# 14.4 Evenness
qiime diversity alpha-group-significance \
  --i-alpha-diversity diversity/evenness_vector.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/evenness-significance.qzv \
  --verbose


################################################################################
# 15. BETA DIVERSITY
################################################################################

# 15.1 Unweighted UniFrac
qiime emperor plot \
  --i-pcoa diversity/unweighted_unifrac_pcoa_results.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/unweighted_unifrac_emperor.qzv \
  --verbose

# 15.2 Weighted UniFrac
qiime emperor plot \
  --i-pcoa diversity/weighted_unifrac_pcoa_results.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/weighted_unifrac_emperor.qzv \
  --verbose

# 15.3 Jaccard
qiime emperor plot \
  --i-pcoa diversity/jaccard_pcoa_results.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/jaccard_emperor.qzv \
  --verbose

# 15.4 Bray-Curtis
qiime emperor plot \
  --i-pcoa diversity/bray_curtis_pcoa_results.qza \
  --m-metadata-file sample-metadata.tsv \
  --o-visualization diversity/bray_curtis_emperor.qzv \
  --verbose

#### END OF QIIME2 PROCESSING ####
